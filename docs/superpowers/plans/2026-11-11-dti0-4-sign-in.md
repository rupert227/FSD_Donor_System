> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
>
> **Working agreement for this project:** the developer (Leon) is new to React, Next.js and TypeScript and wants to learn the code. Before each task, explain what it does and wait for his OK. Keep each change small. After each task, explain what changed and ask one short check question. Do not touch files outside the current task.

**Goal:** An FSD officer signs in with email and password (Supabase Auth) and can only reach the modules their role allows.

**Architecture:** Supabase Auth holds passwords and sessions. `user_accounts` is linked to the Supabase user by `auth_user_id` and keeps role and status. The role is mirrored into the token (`app_metadata.user_role`) by a SQL seed, so the Next.js proxy can check sign-in and module access from the token alone, with no database call. The login form is a server action. The navbar loads the user in the browser.

**Tech Stack:** Next.js 16.4.1 (App Router, `cacheComponents: true`, `proxy.ts`), React 19, TypeScript, `@supabase/supabase-js` 2.118 and `@supabase/ssr` 0.12, Tailwind v4, lucide-react. Runs in Docker (Node 22).

**Spec:** `docs/superpowers/specs/2026-11-11-dti0-4-sign-in-design.md`

```ts
// Who can open what. The only place the role-to-module rules live (decision D-8).
// No imports on purpose: the proxy, the login action and the navbar all use it.

export type UserRole =
  | "Conversion Officer"
  | "Retention Officer"
  | "Donor Care Officer"
  | "Division Head"
  | "System Administrator";

export type ModuleKey = "dashboard" | "donor-conv" | "donor-retn" | "donor-care";

// The URL prefix of each module.
const MODULE_PATH: Record<ModuleKey, string> = {
  dashboard: "/dashboard",
  "donor-conv": "/donor-conv",
  "donor-retn": "/donor-retn",
  "donor-care": "/donor-care",
};

// The first page of each module. Matches the module switcher in the navbar.
const MODULE_HOME: Record<ModuleKey, string> = {
  dashboard: "/dashboard",
  "donor-conv": "/donor-conv/prospects",
  "donor-retn": "/donor-retn/donors",
  "donor-care": "/donor-care/transactions",
};

const ALL_MODULES: ModuleKey[] = ["dashboard", "donor-conv", "donor-retn", "donor-care"];

// Which modules each role can open, and which one it lands on after signing in.
const ROLE_ACCESS: Record<UserRole, { modules: ModuleKey[]; home: ModuleKey }> = {
  "Conversion Officer": { modules: ["donor-conv"], home: "donor-conv" },
  "Retention Officer": { modules: ["donor-retn"], home: "donor-retn" },
  "Donor Care Officer": { modules: ["donor-care"], home: "donor-care" },
  "Division Head": { modules: ALL_MODULES, home: "dashboard" },
  "System Administrator": { modules: ALL_MODULES, home: "dashboard" },
};

// Pages anyone can open without signing in.
const PUBLIC_PATHS = ["/login", "/forgot-password", "/auth/callback", "/api/health"];

export function isUserRole(value: unknown): value is UserRole {
  return typeof value === "string" && value in ROLE_ACCESS;
}

export function allowedModules(role: UserRole): ModuleKey[] {
  return ROLE_ACCESS[role].modules;
}

export function canAccess(role: UserRole, module: ModuleKey): boolean {
  return ROLE_ACCESS[role].modules.includes(module);
}

export function homePathFor(role: UserRole): string {
  return MODULE_HOME[ROLE_ACCESS[role].home];
}

// Which module a URL belongs to. Shared pages (notifications, preferences) belong to none.
export function moduleForPath(pathname: string): ModuleKey | undefined {
  return ALL_MODULES.find(
    (key) =>
      pathname === MODULE_PATH[key] || pathname.startsWith(`${MODULE_PATH[key]}/`),
  );
}

export function isPublicPath(pathname: string): boolean {
  return PUBLIC_PATHS.some(
    (path) => pathname === path || pathname.startsWith(`${path}/`),
  );
}
```

```ts
"use server";

import { redirect } from "next/navigation";
import { homePathFor, isUserRole } from "@/lib/access";
import { createClient } from "@/lib/supabase/server";

export type SignInState = { error: string | null };

const WRONG_LOGIN = "Incorrect email or password.";
const NOT_SET_UP = "Your account is not set up yet. Contact the System Administrator.";
const SUSPENDED = "This account is suspended. Contact the System Administrator.";

export async function signIn(
  _previous: SignInState,
  formData: FormData,
): Promise<SignInState> {
  const email = String(formData.get("email") ?? "").trim();
  const password = String(formData.get("password") ?? "");

  if (!email || !password) {
    return { error: "Enter your email address and password." };
  }

  const supabase = await createClient();

  // Same message for a wrong email and a wrong password.
  const { data, error } = await supabase.auth.signInWithPassword({
    email,
    password,
  });
  if (error || !data.user) {
    return { error: WRONG_LOGIN };
  }

  const { data: account } = await supabase
    .from("user_accounts")
    .select("user_role, account_status")
    .eq("auth_user_id", data.user.id)
    .maybeSingle();

  if (!account || !isUserRole(account.user_role)) {
    await supabase.auth.signOut();
    return { error: NOT_SET_UP };
  }

  if (account.account_status === "Suspended") {
    await supabase.auth.signOut();
    return { error: SUSPENDED };
  }

  // redirect() must stay outside try/catch: it works by throwing.
  redirect(homePathFor(account.user_role));
}
```

```ts
import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";
import { homePathFor, isPublicPath, isUserRole } from "@/lib/access";

export async function updateSession(request: NextRequest) {
  let supabaseResponse = NextResponse.next({ request });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value }) =>
            request.cookies.set(name, value),
          );
          supabaseResponse = NextResponse.next({ request });
          cookiesToSet.forEach(({ name, value, options }) =>
            supabaseResponse.cookies.set(name, value, options),
          );
        },
      },
    },
  );

  // Verifies the sign-in token and refreshes it when it has expired. Do not remove.
  // Reads the token only, with no database call, so it stays cheap on every request.
  const { data } = await supabase.auth.getClaims();
  const claims = data?.claims;

  const pathname = request.nextUrl.pathname;

  // A redirect that keeps any refreshed session cookies.
  function redirectTo(path: string) {
    const url = request.nextUrl.clone();
    url.pathname = path;
    url.search = "";
    const response = NextResponse.redirect(url);
    supabaseResponse.cookies
      .getAll()
      .forEach((cookie) => response.cookies.set(cookie));
    return response;
  }

  // Not signed in: only the public pages are open.
  if (!claims) {
    return isPublicPath(pathname) ? supabaseResponse : redirectTo("/login");
  }

  const role = claims.app_metadata?.user_role;

  // Signed in but the account has no role (not set up): only the login page, which explains it.
  if (!isUserRole(role)) {
    if (pathname === "/login") return supabaseResponse;
    const url = request.nextUrl.clone();
    url.pathname = "/login";
    url.search = "?error=no-account";
    return NextResponse.redirect(url);
  }

  // Already signed in: no reason to see the login page again.
  if (pathname === "/login") {
    return redirectTo(homePathFor(role));
  }

  return supabaseResponse;
}
```

```ts
export type ResetState = { message: string | null };

export async function requestReset(
  _previous: ResetState,
```

```ts
import { NextResponse, type NextRequest } from "next/server";
import { createClient } from "@/lib/supabase/server";

// The emailed link lands here with a one-time code. Trade it for a session, then continue.
export async function GET(request: NextRequest) {
  const base = process.env.NEXT_PUBLIC_APP_URL ?? request.nextUrl.origin;
  const code = request.nextUrl.searchParams.get("code");
  const next = request.nextUrl.searchParams.get("next") ?? "/";

  // Only allow a path on this site, never another address.
  const safeNext = next.startsWith("/") && !next.startsWith("//") ? next : "/";

  if (code) {
    const supabase = await createClient();
    const { error } = await supabase.auth.exchangeCodeForSession(code);
    if (!error) {
      return NextResponse.redirect(new URL(safeNext, base));
    }
  }

  return NextResponse.redirect(new URL("/login", base));
}
```

```tsx
"use client";

import { useState, type FormEvent } from "react";
import { createClient } from "@/lib/supabase/client";

const MIN_LENGTH = 8;

export function ResetPasswordForm() {
  const [error, setError] = useState<string | null>(null);
  const [saving, setSaving] = useState(false);

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const form = new FormData(event.currentTarget);
    const password = String(form.get("password") ?? "");
    const confirm = String(form.get("confirm") ?? "");

    if (password.length < MIN_LENGTH) {
      setError(`Use at least ${MIN_LENGTH} characters.`);
      return;
    }
    if (password !== confirm) {
      setError("The two passwords do not match.");
      return;
    }

    setError(null);
    setSaving(true);
    const { error: updateError } = await createClient().auth.updateUser({ password });
    if (updateError) {
      setSaving(false);
      setError("The password could not be changed. Request a new reset link.");
      return;
    }

    // Signed in with the new password: go to the app (the proxy sends them to their home).
    window.location.href = "/";
  }

  const fieldClass =
    "border-border-strong bg-surface-white text-text-primary h-10 w-full rounded-[6px] border px-3 text-[0.9rem]";

  return (
    <form onSubmit={handleSubmit} className="focus-scope flex flex-col gap-4" noValidate>
      <div>
        <h1 className="text-text-primary text-[1.35rem] leading-7 font-semibold">
          Choose a new password
        </h1>
        <p className="text-text-secondary text-[0.85rem]">
          Use at least {MIN_LENGTH} characters.
        </p>
      </div>

      {error && (
        <p
          role="alert"
          className="border-border-warning bg-surface-warning-strong text-text-warning rounded-[6px] border px-3 py-2 text-[0.85rem]"
        >
          {error}
        </p>
      )}

        <label htmlFor="password" className="text-text-primary text-[0.85rem] font-medium">
          New password
        </label>

          name="password"
          type="password"
          autoComplete="new-password"
          required
          className={fieldClass}
        />
      </div>

        <label htmlFor="confirm" className="text-text-primary text-[0.85rem] font-medium">
```
